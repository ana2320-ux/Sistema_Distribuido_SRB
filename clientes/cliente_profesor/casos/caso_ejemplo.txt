# Caso de prueba de ejemplo (formato para el caso del profesor)
#
# cliente  OPERACION  datos                          => resultado esperado (opcional)
#
# RESERVAR  evento ocurrencia asientos
# MODIFICAR evento ocurrencia_actual ocurrencia_nueva asientos
# CANCELAR  evento ocurrencia asientos
# (ocurrencia = 1 o 2: la 1.a o 2.a fecha del evento en el mes)
#
# Primera entrega: solo CONSULTAR y RESERVAR.

C1  CONSULTAR
C1  RESERVAR  E5   1  4                 => OK
C2  RESERVAR  E5   1  96                => OK
C3  RESERVAR  E5   1  1                 => RECHAZADO    # E5 fecha 1 ya está llena
C3  RESERVAR  E5   2  10                => OK           # E5 tiene una segunda fecha
C2  RESERVAR  E30  1  50                => OK           # E30 solo tiene una fecha
C1  RESERVAR  E30  1  51                => RECHAZADO    # quedan 50
